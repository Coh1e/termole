cmake_minimum_required(VERSION 3.28)
project(termole_native_primitives LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows is required.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(WIL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WIL_BUILD_PACKAGING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(wil GIT_REPOSITORY https://github.com/microsoft/wil.git
  GIT_TAG a1c0cbe1d7065aef6333399d3e25608eacdffc0f)
FetchContent_MakeAvailable(wil)
add_executable(native_view_probe native_view.cpp)
target_compile_definitions(native_view_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(native_view_probe PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(native_view_probe PRIVATE WIL d2d1 dwrite windowscodecs ole32 user32)
enable_testing()
add_test(NAME native_primitives COMMAND native_view_probe "${CMAKE_CURRENT_BINARY_DIR}/evidence/native-text.png")
set_tests_properties(native_primitives PROPERTIES TIMEOUT 20)
