cmake_minimum_required(VERSION 3.28)
project(termole_picker_spike LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This spike requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(WIL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WIL_BUILD_PACKAGING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(wil
  GIT_REPOSITORY https://github.com/microsoft/wil.git
  GIT_TAG a1c0cbe1d7065aef6333399d3e25608eacdffc0f
  GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(wil)
add_executable(picker_spike picker_spike.cpp)
target_link_libraries(picker_spike PRIVATE WIL)
target_compile_definitions(picker_spike PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(picker_spike PRIVATE /W4 /permissive- /utf-8)
enable_testing()
set(TERMOLE_FZF_EXECUTABLE "" CACHE FILEPATH "Unmodified Windows fzf executable")
if(TERMOLE_FZF_EXECUTABLE)
  add_test(NAME picker_channels COMMAND picker_spike "${TERMOLE_FZF_EXECUTABLE}" "${CMAKE_CURRENT_BINARY_DIR}/evidence")
  set_tests_properties(picker_channels PROPERTIES TIMEOUT 60)
endif()
