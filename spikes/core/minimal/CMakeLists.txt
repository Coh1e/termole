cmake_minimum_required(VERSION 3.25)
project(termole_core_minimal VERSION 0.6.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_compile_definitions(NOMINMAX _USE_MATH_DEFINES)
add_compile_options(/utf-8 /bigobj)
include(FetchContent)
set(TERM_LOCAL "${CMAKE_CURRENT_LIST_DIR}/../../../.local")
set(FETCHCONTENT_BASE_DIR "${TERM_LOCAL}/fetchcontent-core" CACHE PATH "Project-local dependency cache")
set(ENABLE_TIDY OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_TESTING OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_TOOLS OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBUNICODE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(GSL_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(gsl GIT_REPOSITORY https://github.com/microsoft/GSL.git GIT_TAG 0f6dbc9e2915ef5c16830f3fa3565738de2a9230 GIT_SUBMODULES "")
FetchContent_Declare(boxed GIT_REPOSITORY https://github.com/contour-terminal/boxed-cpp.git GIT_TAG 82b362155c4e80d274282b51f1e4750af27e471a GIT_SUBMODULES "")
FetchContent_Declare(reflection GIT_REPOSITORY https://github.com/contour-terminal/reflection-cpp.git GIT_TAG 8a83b3259f1be01098f942d525bcead0cfd14a74 GIT_SUBMODULES "")
FetchContent_Declare(unicode GIT_REPOSITORY https://github.com/contour-terminal/libunicode.git GIT_TAG 25648e888e2433604c3e6f392796910f761d2285 GIT_SUBMODULES "")
FetchContent_Declare(contour GIT_REPOSITORY https://github.com/contour-terminal/contour.git GIT_TAG 3aef6f7243f571fd4fda2642da190a244bb87a57 GIT_SUBMODULES "" SOURCE_SUBDIR termole-no-auto-build)
FetchContent_MakeAvailable(contour gsl boxed reflection unicode)
set(CONTOUR_TESTING OFF)
set(CONTOUR_VERSION_STRING "0.6.3-termole-spike")
# Consume only upstream core directories; no frontend, font shaping, SSH or OpenConsole packaging.
set(PROJECT_SOURCE_DIR "${contour_SOURCE_DIR}")
set(CONTOUR_SRC "${PROJECT_SOURCE_DIR}/src")
add_subdirectory("${CONTOUR_SRC}/crispy" crispy)
add_library(vtpty STATIC "${CONTOUR_SRC}/vtpty/MockPty.cpp" "${CONTOUR_SRC}/vtpty/MockViewPty.cpp")
target_include_directories(vtpty PUBLIC "${CONTOUR_SRC}")
target_link_libraries(vtpty PUBLIC crispy::core Microsoft.GSL::GSL)
add_subdirectory("${CONTOUR_SRC}/vtparser" vtparser)
add_subdirectory("${CONTOUR_SRC}/vtbackend" vtbackend)

add_executable(termole_core_probe main.cpp)
target_link_libraries(termole_core_probe PRIVATE vtbackend)
enable_testing()
add_test(NAME core_probe COMMAND termole_core_probe)
